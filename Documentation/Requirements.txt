function command = sequence(battery,dcLink,isolation,crash,drive,charge,mainFeedback,prechargeFeedback,valid,reset)
% Command states: 0=open, 1=precharge, 2=main closed, 3=latched fault.
% Synthetic controller requirement: de-energize on request removal or reset;
% isolation/crash faults latch. Precharge has a 500 ms completion deadline.
persistent state elapsed
if isempty(state), state=0; elapsed=0; end
requested=(drive~=0 || charge~=0);
if reset~=0 || ~requested
 state=0;elapsed=0;
elseif ~valid || isolation<100000 || crash~=0
 state=3;elapsed=0;
elseif state==0
 state=1;elapsed=0;
elseif state==1
 if dcLink>=0.95*battery && prechargeFeedback~=0
  state=2;elapsed=0;
 elseif elapsed>=0.5
  state=3;elapsed=0;
 else
  elapsed=elapsed+SAMPLE_TIME;
 end
elseif state==2 && mainFeedback==0
 state=3;elapsed=0;
end
command=state;
end
